# Versions pinned to the ones actually validated in the field (Pi 5, Debian 13
# trixie, Python 3.13) as of 14/07/2026 -- before, no version was fixed and
# setup_agent.sh performs a destructive `rm -rf .venv` before every
# reinstallation: replacing a Pi several months later could have installed
# incompatible versions (the ultralytics API, .track()/persist, Results formats
# and so on can change between major releases).
#
# numpy is not listed here: it is provided by the system package (apt) thanks to
# `python3 -m venv --system-site-packages` in setup_agent.sh (required for
# picamera2/libcamera) -- numpy==2.2.4 as of 14/07/2026 on the reference Pi;
# psutil and pyserial are pinned below anyway, so as not to depend solely on
# that flag should the venv creation ever change.
# paho-mqtt removed: a dependency used nowhere in the code (no current MQTT use).

# ncnn is the inference backend actually used in production (both models are
# *_ncnn_model exports). It was not listed: Ultralytics then ran `pip install
# ncnn` on the fly at the first model load, which forced a download over 4G on
# every cold start and made initialisation fail whenever the link was not up yet.
# An official cp313/aarch64 wheel (manylinux_2_28) is available for the Pi 5.
# To avoid pip resolving its declared dependencies (numpy, tqdm, requests,
# portalocker, unbounded opencv-python), install it separately:
#   pip install --no-deps ncnn==1.0.20260526
ncnn==1.0.20260526

# lap: the linear assignment solver used by ByteTrack's association, hence by
# RiverTracker. Same problem as ncnn: absent from this file, Ultralytics
# downloaded it on the fly at the first import of the trackers, warning that the
# process had to be restarted for the update to take effect.
# Observed in real conditions on 01/10/2026 during the first commissioning.
lap==0.5.13

opencv-python==5.0.0.93
psutil==7.0.0
ultralytics==8.4.90
pandas==3.0.3
paramiko==5.0.0
onnx==1.22.0
onnxruntime==1.27.0
pyserial==3.5
